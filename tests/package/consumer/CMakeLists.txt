cmake_minimum_required(VERSION 3.25)
project(PackageConsumer LANGUAGES CXX)
enable_testing()
if(EXPECT_MSVC AND NOT CMAKE_CXX_COMPILER_ID STREQUAL "MSVC")
  message(FATAL_ERROR "Visual Studio consumer requires cl.exe")
endif()
find_package(Isoduo REQUIRED COMPONENTS Core)
if(EXPECT_TOOLSET AND NOT Isoduo_MSVC_TOOLSET_VERSION STREQUAL EXPECT_TOOLSET)
  message(FATAL_ERROR "Expected package toolset ${EXPECT_TOOLSET}, found '${Isoduo_MSVC_TOOLSET_VERSION}'")
endif()
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/core-$<CONFIG>.txt" CONTENT "$<TARGET_FILE:Isoduo::Core>")
add_subdirectory("${EXAMPLE}" example)
if(NOT TARGET profile_alias OR (EXPECT_DESKTOP AND NOT TARGET notices_smoke) OR (NOT EXPECT_DESKTOP AND TARGET notices_smoke))
  message(FATAL_ERROR "Consumer targets do not match installed components")
endif()
